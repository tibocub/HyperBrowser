# i2pd 3.0.0 as one statically linked Node-API addon. Build: build.cmd (Windows), build.sh (Linux, macOS).
# Windows links a tiny import lib with only the napi_* calls used (NODE_LIB, from napi.def): node.lib would clash
# with our OpenSSL. Linux and macOS leave napi_* to the host exe.
cmake_minimum_required(VERSION 3.20)
project(i2pd_addon CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded")
set(CMAKE_PREFIX_PATH ${DEPS})
set(Boost_USE_STATIC_LIBS ON)
set(OPENSSL_USE_STATIC_LIBS ON)
find_package(Boost REQUIRED COMPONENTS atomic filesystem program_options)
find_package(OpenSSL REQUIRED)
find_package(ZLIB REQUIRED)
find_package(miniupnpc CONFIG REQUIRED)
file(GLOB I2P_SRC ${I2PD_SRC}/libi2pd/*.cpp ${I2PD_SRC}/libi2pd_client/*.cpp ${I2PD_SRC}/i18n/*.cpp)
list(APPEND I2P_SRC ${I2PD_SRC}/daemon/UPnP.cpp) # the only daemon file: UPnP, started by addon.cpp
add_library(i2pd_addon SHARED addon.cpp ${I2P_SRC})
set_target_properties(i2pd_addon PROPERTIES PREFIX "" SUFFIX ".node")
target_include_directories(i2pd_addon PRIVATE ${I2PD_SRC}/libi2pd ${I2PD_SRC}/libi2pd_client ${I2PD_SRC}/i18n ${I2PD_SRC}/daemon ${NODE_INC})
target_compile_definitions(i2pd_addon PRIVATE NO_TORRENTS BOOST_ASIO_DISABLE_CONCEPTS USE_UPNP MINIUPNP_STATICLIB)
target_link_libraries(i2pd_addon PRIVATE Boost::atomic Boost::filesystem Boost::program_options OpenSSL::SSL OpenSSL::Crypto ZLIB::ZLIB miniupnpc::miniupnpc)
if(WIN32)
  target_compile_definitions(i2pd_addon PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX _WIN32_WINNT=0x0601)
  target_compile_options(i2pd_addon PRIVATE /EHsc /utf-8 /bigobj /MP)
  target_link_libraries(i2pd_addon PRIVATE ${NODE_LIB} delayimp ws2_32 iphlpapi crypt32 mswsock bcrypt)
  target_link_options(i2pd_addon PRIVATE /DELAYLOAD:node.exe)
elseif(APPLE)
  target_link_options(i2pd_addon PRIVATE -undefined dynamic_lookup)
else()
  set_target_properties(i2pd_addon PROPERTIES POSITION_INDEPENDENT_CODE ON)
  target_link_options(i2pd_addon PRIVATE -static-libstdc++ -static-libgcc)
endif()
